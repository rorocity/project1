// --- [1] 초등학생 눈높이에 맞춘 4개 모서리별 규칙 풀(Pool) 정의 --- // 왼쪽 위 규칙 후보 (간단한 수열 규칙) const topLeftRules = [ { name: '10씩 커짐', getVal: (n) => n * 10, desc: '10씩 커지는 규칙' }, { name: '5씩 커짐', getVal: (n) => n * 5, desc: '5씩 커지는 규칙' }, { name: '2의 거듭제곱(2배씩)', getVal: (n) => Math.pow(2, n), desc: '앞 수의 2배씩 커지는 규칙' }, { name: '100씩 커짐', getVal: (n) => n * 100, desc: '100씩 커지는 규칙' } ]; // 오른쪽 위 규칙 후보 (단어 또는 알파벳 반복) const topRightRules = [ { name: '자연보호', getVal: (n) => ['자', '연', '보', '호', '실', '천'][n - 1], desc: "'자연보호실천' 글자 순서" }, { name: '사계절', getVal: (n) => ['봄', '여', '름', '가', '을', '겨'][n - 1], desc: "'봄여름가을겨(울)' 순서" }, { name: 'A-B 반복', getVal: (n) => (n % 2 === 1 ? 'A' : 'B'), desc: 'A와 B가 번갈아 나오는 규칙' }, { name: '무지개색', getVal: (n) => ['빨', '주', '노', '초', '파', '남'][n - 1], desc: '무지개 색깔 순서' } ]; // 왼쪽 아래 규칙 후보 (도형 및 한글 자음) const bottomLeftRules = [ { name: '기본 도형', getVal: (n) => ['○', '△', '□', '⬠', '⬡', '★'][n - 1], desc: '변의 수가 하나씩 늘어나는 도형' }, { name: '한글 자음순', getVal: (n) => ['ㄱ', 'ㄴ', 'ㄷ', 'ㄹ', 'ㅁ', 'ㅂ'][n - 1], desc: '한글 자음 순서 (ㄱ, ㄴ, ㄷ, ㄹ...)' }, { name: '화살표 회전', getVal: (n) => ['↑', '→', '↓', '←', '↑', '→'][n - 1], desc: '시계 방향으로 90도씩 회전하는 화살표' } ]; // 오른쪽 아래 규칙 후보 (간단한 구구단 및 배수) const bottomRightRules = [ { name: '2단', getVal: (n) => n * 2, desc: '2의 배수 (가운데 숫자의 2배)' }, { name: '3단', getVal: (n) => n * 3, desc: '3의 배수 (구구단 3단)' }, { name: '4단', getVal: (n) => n * 4, desc: '4의 배수 (구구단 4단)' }, { name: '9에서 빼기', getVal: (n) => 10 - n, desc: '합이 10이 되는 짝꿍수' } ]; // --- [2] 새로고침 시 무작위로 하나의 규칙 세트를 뽑아 주사위 면 데이터 생성 --- function generateRandomDice() { const pick = (arr) => arr[Math.floor(Math.random() * arr.length)]; const tlRule = pick(topLeftRules); const trRule = pick(topRightRules); const blRule = pick(bottomLeftRules); const brRule = pick(bottomRightRules); // 물음표로 숨길 면 (기본 4번 면) const hiddenFaceNum = 4; const faces = []; for (let n = 1; n <= 6; n++) { faces.push({ center: String(n), tl: String(tlRule.getVal(n)), tr: String(trRule.getVal(n)), bl: String(blRule.getVal(n)), br: String(brRule.getVal(n)), isSecret: (n === hiddenFaceNum) }); } // 퀴즈 정답 팝업에 들어갈 동적 설명문 const explanation = { center: `${hiddenFaceNum} (주사위 번호 순서)`, tl: `${tlRule.getVal(hiddenFaceNum)} (${tlRule.desc})`, tr: `${trRule.getVal(hiddenFaceNum)} (${trRule.desc})`, bl: `${blRule.getVal(hiddenFaceNum)} (${blRule.desc})`, br: `${brRule.getVal(hiddenFaceNum)} (${brRule.desc})` }; return { faces, explanation }; } // 실행: 주사위 데이터와 정답 설명 가져오기 const currentDice = generateRandomDice(); const diceFaces = currentDice.faces; const currentExplanation = currentDice.explanation;
